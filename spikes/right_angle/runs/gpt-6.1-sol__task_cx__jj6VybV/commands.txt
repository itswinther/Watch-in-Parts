import numpy as np, trimesh, manifold3d as md, math, json, os, time
from pathlib import Path
OUT=Path('submission/parts');OUT.mkdir(parents=True,exist_ok=True)
def cyl(r,z,h,x=0,y=0,n=128):return md.Manifold.cylinder(h,r,circular_segments=n).translate((x,y,z))
def box(d,p):return md.Manifold.cube(d).translate(p)
def union(xs):return md.Manifold.batch_boolean(xs,md.OpType.Add)
def save(name,s):
 m=s.to_mesh();tm=trimesh.Trimesh(np.array(m.vert_properties)[:,:3],np.array(m.tri_verts),process=False);tm.export(OUT/(name+'.stl'));print(name,s.volume(),s.num_tri(),len(s.decompose()),flush=True)
def load(name):
 t=trimesh.load(OUT/(name+'.stl'));return md.Manifold(md.Mesh(np.float32(t.vertices),np.uint32(t.faces)))
def gear(n,r,h,back=0):
 # involute flanks, with circular root and tip
 mod=2*r/n; rb=r*math.cos(math.radians(20)); ra=r+mod; rf=r-1.25*mod
 inv=lambda rr: math.sqrt(max(0,(rr/rb)**2-1))-math.acos(min(1,rb/rr))
 half=math.pi/(2*n)+inv(r)-back/r
 p=[]
 for i in range(n):
  c=2*math.pi*i/n
  p.append((rf*math.cos(c-math.pi/n),rf*math.sin(c-math.pi/n)))
  for rr in np.linspace(max(rb,rf),ra,16):
   a=c-half+inv(rr);p.append((rr*math.cos(a),rr*math.sin(a)))
  for a in np.linspace(c-half+inv(ra),c+half-inv(ra),6)[1:]:p.append((ra*math.cos(a),ra*math.sin(a)))
  for rr in np.linspace(ra,max(rb,rf),16)[1:]:
   a=c+half-inv(rr);p.append((rr*math.cos(a),rr*math.sin(a)))
  p.append((rf*math.cos(c+half),rf*math.sin(c+half)))
 return md.CrossSection([p]).extrude(h)
def rz(s,a,c=(-3.6,0,0)):return s.translate(tuple(-v for v in c)).rotate((0,0,a)).translate(c)
def rx(s,a):return s.translate((0,0,3.9)).rotate((a,0,0)).translate((0,0,-3.9))
def carve(blank,driver,ratio,sense,n,phase_step,turn,center):
 cuts=[]
 for a in np.linspace(0,360/(n/ratio),round((360/(n/ratio))/phase_step)+1):
  d=turn(driver,a);d=rz(d,-sense*a/ratio,center)
  cut=blank"'^d
  if not cut.is_empty():cuts.append(cut)
 print('"'union envelope',len(cuts),flush=True)
 env=union(cuts)
 print('repeat envelope',env.num_tri(),flush=True)
 env=union([rz(env,i*360/n,center) for i in range(n)])
 result=blank-env
 print('carved',result.num_tri(),len(result.decompose()),flush=True)
 return result
if __name__=='__main__':
 # stem gear along x, tooth zero originally +x transformed to -z
 pin=gear(10,.75,.8).rotate((0,90,0)).translate((-6.25,0,-3.9))
 stem=pin+cyl(.25,0,1) if False else pin
 shaft=cyl(.23,0,12).rotate((0,90,0)).translate((-17,0,-3.9))
 # axial collars provide thrust restraint at two bearing blocks
 collars=union([cyl(.43,0,.18).rotate((0,90,0)).translate((x,0,-3.9)) for x in [-11.48,-10.52]])
 grip=cyl(.65,0,.9).rotate((0,90,0)).translate((-17,0,-3.9))
 stem=pin+shaft+collars+grip
 save('stem',stem)
 blank=cyl(2.52,-3.85,1.01,-3.6)-cyl(1.65,-3.86,.72,-3.6)
 # cut with slight expanded tooth thickness for running clearance
 cutter=gear(10,.75,.804,back=-.0003).rotate((0,90,0)).translate((-6.252,0,-3.9))
 crown=carve(blank,cutter,3,1,30,.15,rx,(-3.6,0,0))
 crown=crown+cyl(.4,-3.15,1.32,-3.6)
 spur=gear(12,.9,.48).translate((-3.6,0,-2.43))
 crown=crown+spur
 save('transfer',crown)
 # conjugate arbor wheel cut from round disk by rotating spur
 arb_blank=cyl(2.85,-2.44,.48)
 cutter=gear(12,.9,.484,back=-.0003).translate((-3.6,0,-2.442))
 arb=carve(arb_blank,cutter,3,-1,36,.12,lambda s,a:rz(s,a), (0,0,0))
 arb=arb-box((.602,.602,.7),(-.301,-.301,-2.55))
 save('ratchet',arb)
PY
OPENBLAS_NUM_THREADS=1 python scripts/build.py"
