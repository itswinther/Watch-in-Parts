import numpy as np
from scipy.optimize import minimize
from pathlib import Path
P=2*np.pi/30

def envelope(n=481):
 phis=np.linspace(-P/2,P/2,n)
 vals=[]; sols=[]
 prev=None
 for i,phi in enumerate(phis):
  def calc(v):
   a,t=v; q=.1*(1+.3*np.cos(3*(t-a)))
   y=q*np.cos(t)
   return q*np.sin(t)/np.sqrt(1+y*y), a/10-np.arctan(y)-phi
  def obj(v):return -calc(v)[0]
  def con(v):return calc(v)[1]
  best=None
  seeds=[(10*phi, t) for t in [0.4,1.,1.57,2.2,2.8]]
  if prev is not None:seeds.insert(0,prev)
  for seed in seeds:
   r=minimize(obj,seed,method='SLSQP',constraints={'type':'eq','fun':con},bounds=[(10*phi-1.4,10*phi+1.4),(0,np.pi)],options={'ftol':1e-13,'maxiter':100})
   if r.success and (best is None or r.fun<best.fun):best=r
  if best is None: raise RuntimeError(i)
  vals.append(-best.fun);prev=best.x;sols.append(best.x)
 np.savez('scripts/envelope.npz',phi=phis,h=vals,sol=sols)
 print('envelope',min(vals),max(vals),'period endpoints',vals[0],vals[-1],flush=True)
 return phis,np.array(vals)
if __name__=='__main__':envelope()
PY

# Winding works generation and verification

Run from "'`/app`, using the installed Python environment:

1. `python scripts/profile.py` creates the 12-tooth involute pinion section.
2. `python scripts/envelope_polygon.py` computes the conjugate wheel envelope at a 10:1 ratio.
3. `python scripts/build.py` exports the three binary STL solids and `submission/winding.json`.
4. `python scripts/check.py` checks the final files and saves `scripts/check_results.json`.

The wheel envelope is computed above the stem and reflected below it. This gives negative arbor rotation about +z for positive stem rotation about +x. The wheel has 120 repeated teeth. Its integral hub clears the supplied arbor flange and has a 0.602 mm square socket from z=-2.499 to -1.92 mm. The stem has an integral pinion, crown, and thrust collar. A single bridge provides two journal regions and the collar pocket, with two arms ending 0.02 mm below the movement plate.

`quick_mesh.py` checks additional phases within one 30-degree pinion tooth cycle. `diagnose.py` and `contact_location.py` were used to locate numerical/profile errors during development. `envelope.py` is the abandoned initial three-lobed profile experiment; the final generator uses `envelope_polygon.py`.
