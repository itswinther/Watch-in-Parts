import numpy as np, manifold3d as M, trimesh, json, time
from pathlib import Path
OUT=Path('submission/parts'); OUT.mkdir(exist_ok=True,parents=True)
N=12; Z=108; R=.8; C=7.2

def union(xs): return M.Manifold.batch_boolean(xs,M.OpType.Add)
def box(lo,hi): return M.Manifold.cube(np.array(hi)-lo).translate(lo)
def cyl(r,z0,z1,n=128): return M.Manifold.cylinder(z1-z0,r,r,n).translate((0,0,z0))
def export(s,name):
 mesh=s.to_mesh(); t=trimesh.Trimesh(np.array(mesh.vert_properties)[:,:3],np.array(mesh.tri_verts),process=False); t.export(OUT/(name+'.stl')); print(name,'vol',s.volume(),'tri',len(t.faces),'shells',len(s.decompose()),flush=True)
def pinion_profile():
 pitch=2*np.pi/N; rp=R; rb=rp*np.cos(np.deg2rad(20)); rr=rp-1.25*(2*rp/N); ra=rp+2*rp/N
 inv=lambda t: np.tan(t)-t
 ip=inv(np.deg2rad(20)); half=np.pi/(2*N)
 pts=[]
 for k in range(N):
  center=k*pitch
  def pt(r,a): pts.append([r*np.cos(a+center),r*np.sin(a+center)])
  ab=half+ip
  pt(rr,-pitch/2);pt(rr,-ab)
  for r in np.linspace(rb,ra,24):
   a=half+ip-inv(np.arccos(rb/r));pt(r,-a)
  at=half+ip-inv(np.arccos(rb/ra))
  for a in np.linspace(-at,at,9)[1:]:pt(ra,a)
  for r in np.linspace(ra,rb,24)[1:]:
   a=half+ip-inv(np.arccos(rb/r));pt(r,a)
  pt(rr,ab);pt(rr,pitch/2)
 return M.CrossSection([pts])
prof=pinion_profile()
# Profile local xy becomes yz, extrusion local z becomes positive x.
def alongx(s): return s.rotate((0,90,0))
pinion=alongx(prof.extrude(1.3)).translate((C-.65,0,-3.9))
export(pinion,'pinion_test')
# Crown tooth periodic sector. Generate its conjugate by cutting every position
# of a slightly enlarged pinion, expressed in the crown's frame.
a=np.pi/Z
angles=np.linspace(-a,a,12)
pts=[[0,0]]+[[8.0*np.cos(t),8.0*np.sin(t)] for t in angles]
wedge=M.CrossSection([pts]).extrude(1.5).translate((0,0,-3.5))
blank=(cyl(7.75,-3.5,-2.05,432)-cyl(6.65,-3.6,-1.9,432)) "'^ wedge
cutter=alongx(prof.offset(.0005).extrude(1.40)).translate((C-.70,0,-3.9))
print('"'start cutting',flush=True); start=time.time()
# 0.4 degree sampling: chord error less than 0.00001 mm.
cuts=[cutter.rotate((0,0,0))]
for phi in np.arange(-100,100.0001,.4):
 # rotate about the stem axis, then about the arbor in the stationary crown frame
 cut=cutter.translate((0,0,3.9)).rotate((phi,0,0)).translate((0,0,-3.9)).rotate((0,0,phi/9))
 cuts.append(cut)
tooth=M.Manifold.batch_boolean([blank]+cuts,M.OpType.Subtract)
print('tooth',time.time()-start,tooth.volume(),tooth.num_tri(),flush=True)
ring=union([tooth.rotate((0,0,k*360/Z)) for k in range(Z)])
# Broad web and square seat sit above the arbor flange.
web=cyl(6.8,-2.48,-2.05,432)
seat=box([-.302,-.302,-2.6],[.302,.302,-1.8])
wheel=(ring+web)-seat
export(wheel,'ratchet')
# Stem and retaining shoulders; pinion is one solid with the shaft.
shaft=alongx(cyl(.28,0,28.4)).translate((-17.3,0,-3.9))
shoulders=[]
for x in [-11.54,-10.46,9.76,10.84]:
 shoulders.append(alongx(cyl(.52,0,.20)).translate((x,0,-3.9)))
grip=alongx(cyl(.83,0,1.5)).translate((-17.3,0,-3.9))
stem=union([shaft,pinion,grip]+shoulders)
export(stem,'winding_stem')
# Two individually anchored bearing blocks. Their .03 radial and .04 axial gaps
# permit rotation while stopping all six tested translations.
for name,x in [('outer_bearing',-11.0),('inner_bearing',10.3)]:
 block=box([x-.30,-.75,-4.65],[x+.30,.75,-1.72])
 hole=alongx(cyl(.31,0,1)).translate((x-.5,0,-3.9))
 export(block-hole,name)
(OUT/'pinion_test.stl').unlink()
data={'mobiles':{'stem':{'point':[0,0,-3.9],'direction':[1,0,0],'parts':['winding_stem']},'arbor':{'parts':['ratchet']}},'fixed':['outer_bearing','inner_bearing'],'meshes':[{'driver':'stem','driver_teeth':12,'driven':'arbor','driven_teeth':108,'sense':'opposite'}]}
Path('submission/winding.json').write_text(json.dumps(data,indent=2))
PY
OPENBLAS_NUM_THREADS=1 python scripts/build.py"
